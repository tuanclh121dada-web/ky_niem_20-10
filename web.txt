  <script>
    // 1. Đồng hồ đếm thời gian từ 01/06/2026
    const startDate = new Date("2026-06-01T00:00:00");
    function updateCounter() {
      const now = new Date();
      const diff = Math.max(0, now - startDate);

      const days = Math.floor(diff / (1000 * 60 * 60 * 24));
      const hours = Math.floor((diff / (1000 * 60 * 60)) % 24);
      const minutes = Math.floor((diff / 1000 / 60) % 60);
      const seconds = Math.floor((diff / 1000) % 60);

      document.getElementById("days-count").innerText = days;
      document.getElementById("hours-count").innerText = String(hours).padStart(2, "0");
      document.getElementById("minutes-count").innerText = String(minutes).padStart(2, "0");
      document.getElementById("seconds-count").innerText = String(seconds).padStart(2, "0");
    }
    setInterval(updateCounter, 1000);
    updateCounter();

    // 2. Mở phong thư & gõ chữ
    const letterMessage = `Chúc mừng ngày 20/10 công chúa nhỏ của anh! 🌸
Từ ngày 1/6 định mệnh ấy đến nay, chúng mình đã cùng nhau đi qua quãng thời gian thật ngọt ngào. Cảm ơn em vì luôn dịu dàng, đáng yêu và ở bên cạnh anh.

Chúc em ngày 20/10 luôn xinh đẹp rạng rỡ, ngập tràn niềm vui và mãi nắm chặt tay anh trên chặng đường phía trước nhé!`;

    let isOpened = false;
    function openLetter() {
      document.getElementById("envelope-closed").style.display = "none";
      document.getElementById("letter-content").classList.remove("hidden");

      confetti({ particleCount: 90, spread: 70, origin: { y: 0.6 } });

      if (!isOpened) {
        isOpened = true;
        let el = document.getElementById("typewriter-text");
        let i = 0;
        let timer = setInterval(() => {
          if (i < letterMessage.length) {
            el.innerHTML += letterMessage.charAt(i);
            i++;
          } else {
            clearInterval(timer);
          }
        }, 30);
      }
    }

    // 3. Tải ảnh lên các khung Polaroid
    function loadPhoto(event, index) {
      const file = event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(e) {
        const img = document.getElementById(`img-preview-${index}`);
        const placeholder = document.getElementById(`placeholder-${index}`);
        img.src = e.target.result;
        img.classList.remove("hidden");
        placeholder.classList.add("hidden");
      };
      reader.readAsDataURL(file);
    }

    // 4. Sao chép Voucher quà
    function claimVouchers() {
      const checked = Array.from(document.querySelectorAll(".voucher-cb:checked")).map(cb => cb.value);
      const text = `💖 Em đã chốt quà 20/10 gồm: ${checked.join(", ")}! Yêu anh nhiều lắm! 🌸`;
      navigator.clipboard.writeText(text).catch(() => {});
      confetti({ particleCount: 70, spread: 60, origin: { y: 0.7 } });
      const status = document.getElementById("copy-status");
      status.innerText = "✅ Đã sao chép danh sách quà! Em chỉ cần dán (Paste) gửi vào tin nhắn cho anh nhé!";
      status.classList.remove("hidden");
    }

    // 5. Hiệu ứng cánh hoa & trái tim rơi
    const symbols = ["🌸", "💖", "🌷", "✨", "💕"];
    const container = document.getElementById("petals-container");
    for (let i = 0; i < 18; i++) {
      let span = document.createElement("span");
      span.className = "floating-item";
      span.innerText = symbols[Math.floor(Math.random() * symbols.length)];
      span.style.left = Math.random() * 100 + "vw";
      span.style.fontSize = (Math.random() * 14 + 14) + "px";
      span.style.animationDuration = (Math.random() * 6 + 7) + "s";
      span.style.animationDelay = (Math.random() * 5) + "s";
      container.appendChild(span);
    }
